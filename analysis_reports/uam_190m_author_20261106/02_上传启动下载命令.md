## 3. 查看进度、恢复与打包

```bash
cd /home/SY/uam/UAGMC-main
date
pgrep -afu "$(id -u)" "[t]rain_uam_(happo_mappo_dreamer_140m|marie_25m|dima_25m)_ONEFILE[.]py"
for UAM_SUITE in core marie dima; do
    UAM_DIR="serial_runs/uam_${UAM_SUITE}_author_20261106_v1"
    echo "【$UAM_SUITE】"
    for UAM_NAME in progress.json RUNNING.json END.json BUDGET_STOP.json; do
        if [ -f "$UAM_DIR/$UAM_NAME" ]; then cat "$UAM_DIR/$UAM_NAME"; fi
    done
done
tail -n 18 /home/SY/wyw/logs/uam_author_core140m_20261106.log
tail -n 18 /home/SY/wyw/logs/uam_author_marie25m_20261106.log
tail -n 18 /home/SY/wyw/logs/uam_author_dima25m_20261106.log

```

父进程正常结束会自动分包。如果仅需重新打包，不会训练：

```bash
cd /home/SY/uam/UAGMC-main
/home/SY/wyw/conda/envs/uam_server/bin/python -B train_uam_happo_mappo_dreamer_140m_ONEFILE.py --pack-only
/home/SY/wyw/conda/envs/uam_server/bin/python -B train_uam_marie_25m_ONEFILE.py --pack-only
/home/SY/wyw/conda/envs/uam_server/bin/python -B train_uam_dima_25m_ONEFILE.py --pack-only
for UAM_SUITE in core marie dima; do
    for UAM_KIND in DATAONLY FINAL_REPLAY_MODELS; do
        UAM_PACK="experiment_archives/uam_${UAM_SUITE}_author_20261106_v1_${UAM_KIND}.tar.gz"
        tar -tzf "$UAM_PACK" > /dev/null || break
        ls -lh "$UAM_PACK"
        sha256sum "$UAM_PACK"
    done
done
```

中断恢复重新运行对应启动段即可，完整格自动跳过。工程失败日志保留；预算保护停止不能通过盲目重试扩大预算。不要在训练仍进行时打包并把部分包称为完整结果。
